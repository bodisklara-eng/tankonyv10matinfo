# 10. Osztály: Galilei-féle függetlenségi elv (2D Mozgás)
g, m, k, dt = 9.81, 1.0, 0.12, 0.02
x, y, vx, vy, t = 0.0, 0.0, 20.0, 20.0, 0.0 # v0=28.28 m/s, α=45°
while y >= 0:
    v = (vx**2 + vy**2)**0.5 # Eredő sebességvektor hossza
    ax = -(k / m) * v * vx # Horizontális lassulás
    ay = -g - (k / m) * v * vy # Vertikális gyorsulás
    vx += ax * dt; vy += ay * dt
    x += vx * dt; y += vy * dt; t += dt
