1. A Mátrixszorzás Szabálya

Fő szabály: Két mátrixot csak akkor lehet megszorozni egymással, ha az első mátrix oszlopainak száma megegyezik a második mátrix sorainak számával.

  • Az eredmény C mátrix $i$-edik sorának és $j$-edik oszlopának elemét úgy kapjuk meg, hogy az **A** mátrix $i$-edik sorát szorozzuk össze a **B** mátrix $j$-edik oszlopával (elemenként), majd ezeket az értékeket összeadjuk.
  • Módszer neve: "Sor × Oszlop" szabály.

4. Megvalósítás Programkódban

// Mátrixszorzás C++ nyelven (2x2 példa)
#include <iostream>

using namespace std;

int main() {
    int A[2][2] = {{1, 2}, {3, 4}};
    int B[2][2] = {{5, 6}, {7, 8}};
    int C[2][2] = {0};

    // Három egymásba ágyazott ciklus a sor-oszlop szorzáshoz
    for (int i = 0; i < 2; i++) {
        for (int j = 0; j < 2; j++) {
            for (int k = 0; k < 2; k++) {
                C[i][j] += A[i][k] * B[k][j];
            }
        }
    }

    // Eredmény kiírása
    for (int i = 0; i < 2; i++) {
        for (int j = 0; j < 2; j++) {
            cout << C[i][j] << " ";
        }
        cout << endl;
    }
    return 0;
}
# Mátrixszorzás Python nyelven (2x2 példa)

A = [
    [1, 2],
    [3, 4]
]

B = [
    [5, 6],
    [7, 8]
]

# Eredmény mátrix inicializálása nullákkal
C = [[0, 0], [0, 0]]

# Sor-oszlop szorzás ciklusokkal
for i in range(2):
    for j in range(2):
        for k in range(2):
            C[i][j] += A[i][k] * B[k][j]

# Eredmény kiíratása
for row in C:
    print(row)
