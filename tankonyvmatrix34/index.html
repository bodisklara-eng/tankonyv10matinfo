Python Implementáció

Tiszta Python ciklusokkal és NumPy könyvtár használatával

# Python Mátrixszorzás példa

import numpy as np

# 1. Megvalósítás tiszta Python ciklusokkal (beépített listákkal)
def matrix_multiply_custom(A, B):
    rows_A = len(A)
    cols_A = len(A[0])
    rows_B = len(B)
    cols_B = len(B[0])

    # Ellenőrzés: Az A oszlopainak száma meg kell egyezzen B soraival
    if cols_A != rows_B:
        raise ValueError("Nem szorozható össze: A oszlopai != B sorai")

    # Eredmény mátrix inicializálása 0-kkal (rows_A x cols_B)
    C = [[0 for _ in range(cols_B)] for _ in range(rows_A)]

    # Mátrixszorzás 3 egymásba ágyazott ciklussal
    for i in range(rows_A):
        for j in range(cols_B):
            for k in range(cols_A):
                C[i][j] += A[i][k] * B[k][j]

    return C

# 2. Megvalósítás NumPy könyvtárral (ipari sztenderd, optimális)
def matrix_multiply_numpy(A, B):
    np_A = np.array(A)
    np_B = np.array(B)
    return np.dot(np_A, np_B) # Vagy: np_A @ np_B

# Tesztelés
if __name__ == "__main__":
    A = [
        [1, 2, 3],
        [4, 5, 6]
    ]
    
    B = [
        [7, 8],
        [9, 1],
        [2, 3]
    ]

    print("--- Tiszta Python eredmény ---")
    result_custom = matrix_multiply_custom(A, B)
    for row in result_custom:
        print(row)

    print("\n--- NumPy eredmény ---")
    result_np = matrix_multiply_numpy(A, B)
    print(result_np)

C++ Implementáció

Dinamikus méretű 2D std::vector használatával

#include <iostream>
#include <vector>
#include <iomanip>

using Matrix = std::vector<std::vector<int>>;

// Mátrixszorzó függvény C++-ban
Matrix multiplyMatrices(const Matrix& A, const Matrix& B) {
    int rowsA = A.size();
    int colsA = A[0].size();
    int rowsB = B.size();
    int colsB = B[0].size();

    // Ellenőrzés: A oszlopai == B sorai
    if (colsA != rowsB) {
        std::cerr << "Hiba: A mátrixok mérete nem megfelelő a szorzáshoz!" << std::endl;
        return {};
    }

    // Eredmény C mátrix inicializálása (rowsA x colsB) csupa 0 értékkel
    Matrix C(rowsA, std::vector<int>(colsB, 0));

    // Mátrixszorzás elvégzése
    for (int i = 0; i < rowsA; ++i) {
        for (int j = 0; j < colsB; ++j) {
            for (int k = 0; k < colsA; ++k) {
                C[i][j] += A[i][k] * B[k][j];
            }
        }
    }

    return C;
}

// Mátrix kiíratása a konzolra
void printMatrix(const Matrix& mat) {
    for (const auto& row : mat) {
        for (int val : row) {
            std::cout << std::setw(5) << val << " ";
        }
        std::cout << std::endl;
    }
}

int main() {
    // 2x3-as A mátrix
    Matrix A = {
        {1, 2, 3},
        {4, 5, 6}
    };

    // 3x2-es B mátrix
    Matrix B = {
        {7, 8},
        {9, 1},
        {2, 3}
    };

    std::cout << "A Mátrix:" << std::endl;
    printMatrix(A);

    std::cout << "\nB Mátrix:" << std::endl;
    printMatrix(B);

    Matrix C = multiplyMatrices(A, B);

    std::cout << "\nEredmény C Mátrix (A * B):" << std::endl;
    printMatrix(C);

    return 0;
}
