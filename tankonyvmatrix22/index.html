Mátrix feltöltés soronkénti végrehajtással és valós idejű kódkövetéssel (n = 4)

C++ Kód
for(i = 1; i <= n; i++) {
    for(j = 1; j <= n; j++) {
        if(i == j) a[i][j] = 0;
        else if(j == n - i + 1) a[i][j] = 2;
        else a[i][j] = 3;
    }
}
