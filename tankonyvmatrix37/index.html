Interakciós Mód:
Belső Dimenziók Megegyeznek! (A oszlopai = B sorai) A szorzás elvégezhető. Az eredmény mátrix (C) mérete: 2 × 2

Miért kell a belső értéknek megegyeznie? Az eredmény $C_{i,j}$ eleme az $A$ mátrix $i$-edik sorának és a $B$ mátrix $j$-edik oszlopának skaláris szorzata (páronkénti szorzás összege). Ha az $A$ sorában lévő elemek száma (oszlopok) nem egyezik meg a $B$ oszlopában lévő elemek számával (sorok), akkor bizonyos elemeknek nem lenne szorzópárjuk.

Generált algoritmus tetszőleges dimenziójú mátrixok szorzására.

matrix_multiplication.py 
import numpy as np

def matrix_multiply(A, B):
    """
    Tetszőleges dimenziójú mátrixok szorzása 3-as beágyazott ciklussal.
    """
    rows_A = len(A)
    cols_A = len(A[0])
    rows_B = len(B)
    cols_B = len(B[0])

    # 1. Belső dimenziók ellenőrzése
    if cols_A != rows_B:
        raise ValueError(f"Hiba: A belső dimenziók nem egyeznek meg! ({cols_A} != {rows_B})")

    # 2. Eredmény mátrix inicializálása nullákkal (N x L)
    C = [[0 for _ in range(cols_B)] for _ in range(rows_A)]

    # 3. Szorzás elvégzése (Sor-Oszlop skaláris szorzat)
    for i in range(rows_A):          # A sorai
        for j in range(cols_B):      # B oszlopai
            for k in range(cols_A):  # Közös belső dimenzió
                C[i][j] += A[i][k] * B[k][j]

    return C

# --- TESZTELÉS AZ AKTUÁLIS DINAMIKUS ADATOKKAL ---

matrix_multiplication.cpp 
#include <iostream>
#include <vector>
#include <stdexcept>
#include <iomanip>

using Matrix = std::vector<std::vector<int>>;

// Generikus Mátrixszorzó Függvény
Matrix multiplyMatrices(const Matrix& A, const Matrix& B) {
    size_t rows_A = A.size();
    size_t cols_A = A[0].size();
    size_t rows_B = B.size();
    size_t cols_B = B[0].size();

    // Dimenziók ellenőrzése
    if (cols_A != rows_B) {
        throw std::invalid_argument("A belső dimenzióknak meg kell egyezniük!");
    }

    // Eredmény mátrix inicializálása (N x L) nullákkal
    Matrix C(rows_A, std::vector<int>(cols_B, 0));

    // Ciklusok
    for (size_t i = 0; i < rows_A; ++i) {
        for (size_t j = 0; j < cols_B; ++j) {
            for (size_t k = 0; k < cols_A; ++k) {
                C[i][j] += A[i][k] * B[k][j];
            }
        }
    }

    return C;
}

void printMatrix(const Matrix& M) {
    for (const auto& row : M) {
        for (int val : row) {
            std::cout << std::setw(6) << val;
        }
        std::cout << "\n";
    }
}

int main() {
    Matrix A = {
        {1, 2, 3},
        {4, 5, 6}
    };

    Matrix B = {
        {7, 8},
        {9, 1},
        {2, 3}
    };

    try {
        Matrix C = multiplyMatrices(A, B);
        std::cout << "Eredmeny C Matrix:\n";
        printMatrix(C);
    } catch (const std::exception& e) {
        std::cerr << "Hiba: " << e.what() << std::endl;
    }

    return 0;
}
