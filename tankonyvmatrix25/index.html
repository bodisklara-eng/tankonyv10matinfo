Mátrix Összeadás C++ Szimuláció

C++ Forráskód

float a[10][5], b[10][5], c[10][5];
cin >> n >> m;
// A mátrix beolvasása
for(i=1; i<=n; i++) for(j=1; j<=m; j++) cin >> a[i][j];
// B mátrix beolvasása
for(i=1; i<=n; i++) for(j=1; j<=m; j++) cin >> b[i][j];
// Összeadás és kiírás (C = A + B)
for(i=1; i<=n; i++) {
  for(j=1; j<=m; j++) {
    c[i][j] = a[i][j] + b[i][j];
    cout << c[i][j] << " ";
  }
}
